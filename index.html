# Mathematical Poisson & Expected Value Selection Core
def filter_top_10_parlay(matches):
    selected = []
    for match in matches:
        xg_home, xg_away = calculate_expected_goals(match)
        prob_win = poisson_probability(xg_home, xg_away)
        odds = match['market_odds']
        
        # Strict Mathematical Filter Threshold
        if odds >= 1.50 and prob_win >= 0.70:
            expected_value = prob_win * odds
            selected.append({**match, 'value': expected_value})
            
    selected.sort(key=lambda x: x['value'], reverse=True)
    return selected[:10]
